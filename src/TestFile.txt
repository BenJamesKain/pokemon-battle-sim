// Battle goes until one of the pokemon has 0 hp
while(pokemon1.hp>0 && pokemon2.hp>0) {

    // Battle goes on for 100 turns
    for (int i = 0; i <100; i++) {

        // Each of these three lines is a pokemon attacking/performing an action on a target.
        // User - Action - Target
        pokemon1 move1 pokemon2;

        pokemon1 move1 pokemon1;

        pokemon2 move4 pokemon1;

        // Checking if any pokemon died
        if(pokemon1.hp<0) {
            // printing the loser message
            print pokemon1.type + " lost the battle";
            // getting the name of the pokemon who won
            name = pokemon2.name;
            // Exiting the block
            break;

        // Same thing here, but with the other pokemon.
        } elseif(pokemon1.hp<0) {
            print pokemon2.type + " lost the battle";
            name = pokemon1.name;
            break;

        // Both pokemon are alive at the end of the turn
        } else {    

            // Move onto the next turn
            print "Battle continue";
            continue;
        }
        
    }
    
}
